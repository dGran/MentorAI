class, __init__ y self

En Python, self es el equivalente de $this en PHP, pero siempre explícito como primer parámetro de cualquier método de instancia. El constructor se llama __init__, no __construct.

class Cuenta:
    tasa_interes: float = 0.05  # atributo de clase

    def __init__(self, titular: str, saldo: float = 0.0) -> None:
        self.titular = titular  # atributo de instancia
        self._saldo = saldo     # convención: _ indica "protegido"

    @property
    def saldo(self) -> float:
        return self._saldo

    @saldo.setter
    def saldo(self, valor: float) -> None:
        if valor < 0:
            raise ValueError("El saldo no puede ser negativo")
        self._saldo = valor

    def depositar(self, cantidad: float) -> None:
        self._saldo += cantidad

    def __repr__(self) -> str:
        return f"Cuenta(titular={self.titular!r}, saldo={self._saldo:.2f})"


cuenta = Cuenta("Ana", 1000.0)
cuenta.depositar(500.0)
cuenta.saldo = 200.0   # usa el setter
print(cuenta.saldo)    # 200.0
print(cuenta)          # Cuenta(titular='Ana', saldo=200.00)

@property convierte un método en un atributo de lectura. Añadir @nombre.setter permite escritura con validación. En PHP el equivalente sería getSaldo() / setSaldo(); en Python el acceso es sintácticamente idéntico a un atributo público (cuenta.saldo) pero la lógica va en el método.

Atributos de clase vs instancia: tasa_interes es compartida por todas las instancias. Si haces cuenta.tasa_interes = 0.03, creas un atributo de instancia que sombrea el de clase solo para esa instancia. La clase original no cambia.

Métodos de instancia, clase y estáticos

class Cuenta:
    _contador_cuentas: int = 0

    def __init__(self, titular: str) -> None:
        self.titular = titular
        Cuenta._contador_cuentas += 1

    def depositar(self, cantidad: float) -> None:
        self._saldo += cantidad  # método de instancia: accede a self

    @classmethod
    def abrir(cls, titular: str) -> "Cuenta":
        return cls(titular)  # factory method: recibe la clase, no la instancia

    @classmethod
    def total_cuentas(cls) -> int:
        return cls._contador_cuentas

    @staticmethod
    def es_iban_valido(iban: str) -> bool:
        return iban.startswith("ES") and len(iban) == 24


cuenta = Cuenta.abrir("Bob")
print(Cuenta.total_cuentas())  # 1
print(Cuenta.es_iban_valido("ES1234567890123456789012"))  # True

La diferencia entre @classmethod y @staticmethod: el class method recibe la clase como primer argumento (cls) y puede crear instancias, crear subclases correctamente y acceder a atributos de clase. El static method no recibe ni self ni cls; es simplemente una función que vive en el namespace de la clase por organización.

Herencia y clases abstractas

from abc import ABC, abstractmethod


class Animal(ABC):
    def __init__(self, nombre: str) -> None:
        self.nombre = nombre

    @abstractmethod
    def hablar(self) -> str: ...

    def presentarse(self) -> str:
        return f"Soy {self.nombre} y digo: {self.hablar()}"


class Perro(Animal):
    def hablar(self) -> str:
        return "Guau"


class Gato(Animal):
    def hablar(self) -> str:
        return "Miau"


perro = Perro("Rex")
print(perro.presentarse())  # Soy Rex y digo: Guau

Animal("nombre")
# TypeError: Can't instantiate abstract class Animal without hablar

Python permite herencia múltiple. El orden de resolución (MRO) sigue el algoritmo C3, consultable con Clase.__mro__. Para casos simples es predecible; para jerarquías complejas, composición sobre herencia es la recomendación habitual.

class CuentaAhorro(Cuenta):
    def __init__(self, titular: str, tasa: float) -> None:
        super().__init__(titular)  # llama al __init__ de la clase padre
        self._tasa = tasa

@dataclass

Para clases que son principalmente contenedores de datos, @dataclass genera automáticamente __init__, __repr__ y __eq__ a partir de las anotaciones de tipo:

from dataclasses import dataclass, field


@dataclass
class Producto:
    nombre: str
    precio: float
    etiquetas: list[str] = field(default_factory=list)
    activo: bool = True

    def precio_con_iva(self, porcentaje: float = 21.0) -> float:
        return self.precio * (1 + porcentaje / 100)


@dataclass(frozen=True)
class Coordenada:
    latitud: float
    longitud: float


producto = Producto("Teclado", 79.99, ["periférico", "USB"])
print(producto)  # Producto(nombre='Teclado', precio=79.99, ...)
print(producto == Producto("Teclado", 79.99, ["periférico", "USB"]))  # True

coordenada = Coordenada(40.4168, -3.7038)
coordenada.latitud = 0  # FrozenInstanceError: cannot assign to field

frozen=True hace la instancia inmutable (equivalente a readonly en PHP 8.1 o a una struct inmutable en Rust). El objeto se vuelve hashable y puede usarse como clave de diccionario o elemento de set.

Métodos dunder

Los métodos dunder (double underscore) son el protocolo de Python para integrar los objetos con operadores y funciones built-in:

class Vector:
    def __init__(self, x: float, y: float) -> None:
        self.x = x
        self.y = y

    def __repr__(self) -> str:
        return f"Vector({self.x}, {self.y})"

    def __str__(self) -> str:
        return f"({self.x}, {self.y})"

    def __eq__(self, other: object) -> bool:
        if not isinstance(other, Vector):
            return NotImplemented
        return self.x == other.x and self.y == other.y

    def __add__(self, other: "Vector") -> "Vector":
        return Vector(self.x + other.x, self.y + other.y)

    def __len__(self) -> int:
        return 2

    def __getitem__(self, indice: int) -> float:
        if indice == 0:
            return self.x
        if indice == 1:
            return self.y
        raise IndexError(f"Índice {indice} fuera de rango")


v1 = Vector(1, 2)
v2 = Vector(3, 4)
print(v1 + v2)     # (4, 6)
print(v1 == v2)    # False
print(len(v1))     # 2
print(v1[0])       # 1.0

Resumen

  • self es el equivalente de $this pero siempre explícito en la firma del método. __init__ es el constructor.
  • @property expone atributos con lógica de getter/setter sin cambiar la sintaxis de acceso; es el patrón Python en lugar de getX()/setX().
  • @classmethod recibe la clase (no la instancia) y sirve como factory method; @staticmethod es solo organización de namespace.
  • @dataclass genera __init__, __repr__ y __eq__ automáticamente; frozen=True lo hace inmutable y hashable.
  • Los métodos dunder integran los objetos con los operadores y funciones built-in de Python: +, ==, len(), with, indexación, etc.
